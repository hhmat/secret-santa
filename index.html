<script>


function shuffle(array){

    for(let i=array.length-1;i>0;i--){

        const j=Math.floor(Math.random()*(i+1));

        [array[i],array[j]]=[array[j],array[i]];

    }

}


function generate(){


    const names=document.getElementById("names")

        .value

        .split("\n")

        .map(x=>x.trim())

        .filter(Boolean);


    if(names.length<2){

        alert("Add at least two participants.");

        return;

    }


    const exclusionsText=document

        .getElementById("exclusions")

        .value

        .split("\n")

        .filter(Boolean);


    const exclusions=new Set();


    exclusionsText.forEach(line=>{

        const parts=line.split(",");

        if(parts.length===2){


            const a=parts[0].trim();

            const b=parts[1].trim();


            exclusions.add(a+"|"+b);

            exclusions.add(b+"|"+a);

        }

    });


    let assignments=null;

    let attempts=0;


    while(attempts<5000){


        attempts++;


        const recipients=[...names];

        shuffle(recipients);


        let valid=true;


        for(let i=0;i<names.length;i++){


            const giver=names[i];

            const receiver=recipients[i];


            if(giver===receiver){

                valid=false;

                break;

            }


            if(exclusions.has(giver+"|"+receiver)){

                valid=false;

                break;

            }

        }


        if(valid){


            assignments={};


            names.forEach((name,index)=>{

                assignments[name]=recipients[index];

            });


            break;

        }

    }


    if(!assignments){

        alert("Unable to generate assignments. Check exclusions.");

        return;

    }


    showResults(assignments);

}


function showResults(assignments){


    const results=document.getElementById("results");


    results.innerHTML="<h2>Participant Links</h2>";


    Object.keys(assignments).forEach(name=>{


        const recipient=assignments[name];


        const token=btoa(

            encodeURIComponent(JSON.stringify({

                name:name,

                recipient:recipient

            }))

        );


        const revealLink=

            window.location.origin +

            window.location.pathname +

            "#"+token;


        const emailText=

`Subject: Your Secret Santa Assignment


Hello ${name},


Click the private link below to reveal your Secret Santa recipient:


${revealLink}


Please keep it secret.


Happy Secret Santa!`;


        results.innerHTML += `

        <div class="card">


            <h3>${name}</h3>


            <div class="linkBox">${revealLink}</div>


            <details>

                <summary>Email Template</summary>

                <pre>${emailText}</pre>

            </details>


        </div>

        `;

    });

}


window.onload=function(){


    if(window.location.hash.length>1){


        document.body.innerHTML="";


        try{


            const token=

                decodeURIComponent(

                    atob(window.location.hash.substring(1))

                );


            const data=JSON.parse(token);


            document.body.innerHTML=`

            <div class="card">

                <h1>🎁 Secret Santa</h1>


                <p>Hello <b>${data.name}</b></p>


                <p>Your recipient is:</p>


                <h2>${data.recipient}</h2>


                <p>Please keep this secret.</p>

            </div>

            `;

        }

        catch(e){


            document.body.innerHTML=

            "<h2>Invalid Secret Santa Link</h2>";

        }

    }

}
